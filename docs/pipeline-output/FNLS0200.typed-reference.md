# Typed reference implementation of FNLS0200 (generated from the COBOL; it passes the acceptance test)
The pipeline rewrote FNLS0200 mechanically onto typed views of its records: each paragraph is a method (its comment names the paragraph, as the requirements packet's step ids do), each statement a typed assignment that stores with the COBOL rules, and each CALL of FNDTKRN, FNMY0100 is the JDK operation the equivalence probe verified. It reproduces the COBOL program exactly, but it is not yet the design.
Already in package `com.mlcpu.capabilities.fnls0200` and compiled (use them; do not redefine them): the record views LoanRequest, CurrencyRec, MathContextRecord, LoanTerms, Status, DtParmsView, MoneyRequest (typed accessors over the record kit's bytes), FndtkrnOperations, Fnmy0100Operations (the verified operations), and Storage (COBOL store rules for values kept in Java fields).
Your starting files are Fnls0200Program.java and Fnls0200Logic.java: returned as they are, they pass. Refactor them into the design (domain records, enums, services, BigDecimal and LocalDate operations, clear names) and keep every behaviour; edit them with SEARCH/REPLACE blocks or return them whole. The tests judge the behaviour; the design judges the code.

package com.mlcpu.capabilities.fnls0200;

import com.mlcpu.rt.ModernProgram;

/**
 * FNLS0200: every CALL 'FNLS0200' reaches call(parameters); the arguments are viewed as typed records and the
 * program's logic (Fnls0200Logic) runs on them. One instance serves every CALL of the run, so WORKING-STORAGE persists.
 */
public final class Fnls0200Program implements ModernProgram {
    private final Fnls0200Logic logic = new Fnls0200Logic();

    @Override
    public void call(byte[][] parameters) {
        logic.call(LoanRequest.over(parameters[0]), CurrencyRec.over(parameters[1]), MathContextRecord.over(parameters[2]), LoanTerms.over(parameters[3]), Status.over(parameters[4]));
    }
}

package com.mlcpu.capabilities.fnls0200;

import com.mlcpu.rt.ModernStep;
import com.mlcpu.capabilities.fnls0200.records.Pic;
import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * FNLS0200 rewritten onto typed views of its records: each paragraph is a method, each statement a typed assignment that
 * stores with the COBOL rules, and each CALL whose groups the equivalence probe verified is its JDK operation. Generated by
 * the typed-interface projection from the COBOL; it passes the capability's acceptance test (it is the faithful behaviour).
 */
final class Fnls0200Logic {
    /** GOBACK from anywhere in a PERFORM chain: ends the CALL. */
    private static final class Goback extends RuntimeException {
        Goback() {
            super(null, null, false, false);
        }
    }

    private static final Goback GOBACK = new Goback();

    // WORKING-STORAGE: kept between CALLs, as in COBOL.
    /** WS-MONEY PIC S9(13)V9(4) COMP-3 */
    private BigDecimal money = BigDecimal.ZERO.setScale(4);
    /** WS-PCT PIC S9(3)V9(9) COMP-3 */
    private BigDecimal pct = BigDecimal.ZERO.setScale(9);
    /** WS-ABEND-CODE PIC S9(9) COMP */
    private int abendCode = 0;
    /** WS-ABEND-TIMING PIC S9(9) COMP */
    private int abendTiming = 0;
    /** WS-DISPLAY-CODE PIC ZZZ9 */
    private String displayCode = "\0\0\0\0";
    /** DT-PARMS: a WORKING-STORAGE area passed to CALLs. */
    private final DtParmsView dtParms = DtParmsView.create();
    /** LK-MONEY-REQUEST: a WORKING-STORAGE area passed to CALLs. */
    private final MoneyRequest moneyRequest = MoneyRequest.create();
    /** What the last DISPLAY showed (a CALLed program has no console: the abend message carries it). */
    private String console = "";

    private LoanRequest loanRequest;
    private CurrencyRec currencyRec;
    private MathContextRecord mathContext;
    private LoanTerms loanTerms;
    private Status status;

    /** One CALL: the arguments, BY REFERENCE. */
    void call(LoanRequest loanRequest, CurrencyRec currencyRec, MathContextRecord mathContext, LoanTerms loanTerms, Status status) {
        this.loanRequest = loanRequest;
        this.currencyRec = currencyRec;
        this.mathContext = mathContext;
        this.loanTerms = loanTerms;
        this.status = status;
        try {
            main();
        } catch (Goback g) {
            // GOBACK
        }
    }

    /** 0000-MAIN */
    private void main() {
        loanTerms.initialize();
        loanTerms.currencyCode(currencyRec.code());
        loanTerms.decimalPlaces(currencyRec.decimalPlaces());
        loanTerms.multiplesInd(currencyRec.multiplesInd());
        if (loanTerms.isMultiplesPresent()) {
            loanTerms.multiples(currencyRec.multiples());
        } else {
            loanTerms.multiples(0L);
        }
        principal();
        if (status.rc() == Status.Rc.OK) {
            frequencyType();
        }
        if (status.rc() == Status.Rc.OK) {
            builderFields();
            downPayment();
        }
    }

    /** 1000-PRINCIPAL */
    private void principal() {
        money = loanRequest.disbAmount();
        moneyRequest.function(MoneyRequest.Function.OF_DF);
        moneyRequest.site("LAT:580");
        moneyRequest.dfACoef(money.movePointRight(4));
        moneyRequest.dfAExp(-4L);
        callMoney();
        if (status.rc() == Status.Rc.OK) {
            loanTerms.principal(moneyRequest.moneyR());
        }
    }

    /** 2000-FREQUENCY-TYPE */
    private void frequencyType() {
        if ("DAYS".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.DAYS);
        } else if ("WEEKS".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.WEEKS);
        } else if ("MONTHS".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.MONTHS);
        } else if ("YEARS".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.YEARS);
        } else if ("WHOLE_TERM".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.WHOLE_TERM);
        } else if ("INVALID".equals(loanRequest.freqType())) {
            loanTerms.freqType(LoanTerms.FreqType.INVALID);
        } else {
            status.rc(Status.Rc.REJECTED);
            status.rejectCode("RJFREQ01");
            status.rejectSrc("LAT:592");
        }
    }

    /** 3000-BUILDER-FIELDS */
    private void builderFields() {
        loanTerms.numRepayments(loanRequest.numRepayments());
        loanTerms.repayEvery(loanRequest.repayEvery());
        loanTerms.annualRate(loanRequest.annualRate());
        dtParms.function(DtParmsView.Function.DAY_OF_DATE);
        dtParms.yyyymmdd(loanRequest.disbDate());
        callDate();
        loanTerms.expDisbDay(dtParms.result());
        loanTerms.seedDay(dtParms.result());
        dtParms.yyyymmdd(loanRequest.schedStartDate());
        callDate();
        loanTerms.submittedDay(dtParms.result());
        if (loanRequest.daysInMonth() == 1 || loanRequest.daysInMonth() == 30) {
            loanTerms.daysInMonthCode(loanRequest.daysInMonth());
        } else {
            loanTerms.daysInMonth(LoanTerms.DaysInMonth.INVALID);
        }
        if (loanRequest.daysInYear() == 1 || loanRequest.daysInYear() == 360 || loanRequest.daysInYear() == 364 || loanRequest.daysInYear() == 365) {
            loanTerms.daysInYearCode(loanRequest.daysInYear());
        } else {
            loanTerms.daysInYear(LoanTerms.DaysInYear.INVALID);
        }
        loanTerms.fixedLengthInd(loanRequest.fixedLengthInd());
        if (loanTerms.isFixedLengthPresent()) {
            loanTerms.fixedLength(loanRequest.fixedLength());
        }
        loanTerms.downpaySw(loanRequest.downpayEnabled());
        loanTerms.downpayPctInd(loanRequest.downpayPctInd());
        if (loanTerms.isDownpayPctPresent()) {
            loanTerms.downpayPct(loanRequest.downpayPct());
        }
        if ("DECLINING_BALANCE".equals(loanRequest.interestMethod())) {
            loanTerms.interestMethod(LoanTerms.InterestMethod.DECLINING_BALANCE);
        } else if ("FLAT".equals(loanRequest.interestMethod())) {
            loanTerms.interestMethod(LoanTerms.InterestMethod.FLAT);
        } else {
            loanTerms.interestMethod(LoanTerms.InterestMethod.METHOD_INVALID);
        }
        loanTerms.fullTermSw(loanRequest.fullTermTranche());
        loanTerms.intRecogSw(loanRequest.intRecogDisb());
        loanTerms.partialSw(loanRequest.allowPartial());
        loanTerms.lcdIntRecogSw("N");
    }

    /** 4000-DOWN-PAYMENT */
    private void downPayment() {
        loanTerms.downpayAmount(0L);
        if (loanTerms.isDownpayEnabled()) {
            if (loanTerms.principal().signum() > 0 && !loanTerms.isDownpayPctPresent()) {
                status.rc(Status.Rc.REJECTED);
                status.rejectCode("RJNULL01");
                status.rejectSrc("LAT:331");
            } else {
                moneyRequest.function(MoneyRequest.Function.TO_DF);
                moneyRequest.moneyA(loanTerms.principal());
                callMoney();
                moneyRequest.dfA(moneyRequest.dfR());
                pct = loanTerms.downpayPct();
                moneyRequest.dfBCoef(pct.movePointRight(9));
                moneyRequest.dfBExp(-9L);
                moneyRequest.function(MoneyRequest.Function.PERCENTAGE_OF);
                moneyRequest.site("LAT:331");
                callMoney();
                if (status.rc() == Status.Rc.OK) {
                    moneyRequest.dfA(moneyRequest.dfR());
                    moneyRequest.function(MoneyRequest.Function.OF_DF);
                    moneyRequest.site("LAT:330");
                    callMoney();
                    loanTerms.downpayAmount(moneyRequest.moneyR());
                }
            }
        }
    }

    /** 8500-CALL-DATE */
    private void callDate() {
        FndtkrnOperations.call(dtParms.bytes());
        if (dtParms.rc() != 0) {
            abendCode = 9;
            abendProgram();
        }
    }

    /** 8600-CALL-MONEY */
    private void callMoney() {
        Fnmy0100Operations.call(moneyRequest.bytes(), loanTerms.bytes(), mathContext.bytes(), status.bytes());
    }

    /** 9999-ABEND-PROGRAM */
    private void abendProgram() {
        displayCode = Pic.edit(BigDecimal.valueOf(abendCode), "ZZZ9");
        console = "FNLS0200 ABENDING, CODE" + displayCode;
        throw new ModernStep.Abend((int) abendCode, console);
    }
}

/**
 * CALL 'FNDTKRN' as the JDK operations the equivalence probe verified: every traced CALL of each group reproduced byte for
 * byte, then 100000 generated CALLs of the faithful FNDTKRN (the probe's equivalences.json). Generated by the
 * typed-interface projection; a group that is not verified throws instead of guessing.
 * Only the groups of the DT-FUNCTION codes this program passes ('DN') are here.
 */
final class FndtkrnOperations {
    private FndtkrnOperations() {}

    /** CALL 'FNDTKRN' USING DT-PARMS (BY REFERENCE). */
    static void call(byte[] p0) {
        DtParms a0 = DtParms.wrap(p0);
        if (a0.dtFunction().equals("DN")) {
            op1(a0);
            return;
        }
        throw new UnsupportedOperationException("FNDTKRN: no verified operation for DT-FUNCTION " + a0.dtFunction());
    }

    /** DT-FUNCTION = 'DN' (DT-FN-DAY-OF-DATE): verified (traced CALLs 190, all identical; generated 100000, 100000 identical). */
    private static void op1(DtParms a0) {
        LocalDate r;
        try {
            LocalDate dtYyyymmddDate = LocalDate.parse(String.format("%08d", a0.dtYyyymmdd()), DateTimeFormatter.BASIC_ISO_DATE);
            r = dtYyyymmddDate;
        } catch (DateTimeException e) {
            a0.dtResult(0L);
            a0.dtRc(0L);
            return;
        }
        a0.dtResult(ChronoUnit.DAYS.between(LocalDate.of(1600, 12, 31), r));
        a0.dtRc(0L);
    }

    private static ModernStep.Abend abend(int code) {
        return new ModernStep.Abend(code, "FNDTKRN ABENDING, CODE " + code);
    }
}
